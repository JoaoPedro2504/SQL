CREATE TABLE TUTOR (    ID_TUTOR INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,    NOME VARCHAR(100),    ENDERECO VARCHAR(255),    EMAIL VARCHAR(100),    TELEFONE VARCHAR(20));CREATE TABLE PACIENTE (    ID_PACIENTE INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,    NOME VARCHAR(100),    ESPECIE VARCHAR(50),           RACA VARCHAR(50),    PESO DECIMAL(5,2),             IDADE INT,                    DATA_NASCIMENTO DATE,    ID_TUTOR INT,    CONSTRAINT FK_PACIENTE_TUTOR FOREIGN KEY (ID_TUTOR) REFERENCES TUTOR(ID_TUTOR));CREATE TABLE VETERINARIO (    ID_VETERINARIO INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,    NOME VARCHAR(100),    EMAIL VARCHAR(100),    TELEFONE VARCHAR(20),    ESPECIALIDADE VARCHAR(100));CREATE TABLE AGENDAMENTO (    ID_AGENDAMENTO INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,    DATA_AGENDAMENTO DATE,    ID_PACIENTE INT,    ID_VETERINARIO INT,    MOTIVO VARCHAR(255),    CONSTRAINT FK_AGENDAMENTO_PACIENTE FOREIGN KEY (ID_PACIENTE) REFERENCES PACIENTE(ID_PACIENTE),    CONSTRAINT FK_AGENDAMENTO_VETERINARIO FOREIGN KEY (ID_VETERINARIO) REFERENCES VETERINARIO(ID_VETERINARIO));CREATE TABLE RECEITA (    ID_RECEITA INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,    ID_AGENDAMENTO INT,    CONSTRAINT FK_RECEITA_AGENDAMENTO FOREIGN KEY (ID_AGENDAMENTO) REFERENCES AGENDAMENTO(ID_AGENDAMENTO));CREATE TABLE MEDICAMENTO (    ID_MEDICAMENTO INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,    NOME VARCHAR(100),    PRECO DECIMAL(10,2),    DESCRICAO VARCHAR(255));CREATE TABLE ITEM_RECEITA (    ID_ITEM INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,    ID_RECEITA INT,    ID_MEDICAMENTO INT,    CONSTRAINT FK_ITEM_RECEITA_RECEITA FOREIGN KEY (ID_RECEITA) REFERENCES RECEITA(ID_RECEITA),    CONSTRAINT FK_ITEM_RECEITA_MEDICAMENTO FOREIGN KEY (ID_MEDICAMENTO) REFERENCES MEDICAMENTO(ID_MEDICAMENTO)            ALTER TABLE ITEM_RECEITA ADD DOSAGEM VARCHAR(100);ALTER TABLE ITEM_RECEITA
